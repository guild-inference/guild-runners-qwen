cmake_minimum_required(VERSION 3.24)
project(guild_runners_qwen VERSION 1.0.0 LANGUAGES C CXX)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
  OUTPUT_VARIABLE PACKAGE_SOURCE_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
configure_file(package.json.in ${CMAKE_CURRENT_BINARY_DIR}/package.json @ONLY)
find_package(GuildKernelSDK 1.1 REQUIRED CONFIG)
add_library(guild_runners_qwen SHARED src/qwen.cpp)
guild_configure_gkernel_plugin(guild_runners_qwen)
target_compile_features(guild_runners_qwen PRIVATE cxx_std_17)
target_compile_definitions(guild_runners_qwen PRIVATE GUILD_PACKAGE_VERSION="${PROJECT_VERSION}")
set_target_properties(guild_runners_qwen PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(UNIX)
  target_link_options(guild_runners_qwen PRIVATE -Wl,--no-undefined)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(runner_test tests/runner_test.cpp)
  target_link_libraries(runner_test PRIVATE Guild::KernelSDK guild_runners_qwen)
  add_test(NAME runner_contract COMMAND runner_test)
  add_library(c_contract OBJECT tests/c_contract.c)
  target_link_libraries(c_contract PRIVATE Guild::KernelSDK)
endif()
install(TARGETS guild_runners_qwen LIBRARY DESTINATION lib RUNTIME DESTINATION lib)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/package.json DESTINATION . RENAME .gkmetadata)
